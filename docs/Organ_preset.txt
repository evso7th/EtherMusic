// В theremin-processor.js

createOrganVoice(frequency) {
  const voices = [];

  // Основной регистр
  voices.push(this.createOscillator(frequency, 'sine', 0.6));
  voices.push(this.createOscillator(frequency * 2, 'sine', 0.4)); // Октава
  voices.push(this.createOscillator(frequency * 3, 'triangle', 0.3)); // Квинта
  voices.push(this.createOscillator(frequency * 4, 'sawtooth', 0.2)); // Две октавы
  voices.push(this.createOscillator(frequency * 0.5, 'sine', 0.5)); // Субоктава

  // Добавь лёгкий шум (как у воздушного органа)
  if (Math.random() < 0.1) {
    voices.push(this.createNoise(0.02));
  }

  return voices;
}

lfo.frequency.value = 5.5; // Частота вибрато
const gain = this.context.createGain();
gain.gain.value = 2; // Глубина

lfo.connect(gain);
gain.connect(oscillator.frequency);

lfo.start();

Чтобы звук стал "органным", а не "синтезаторным":

filter.frequency.value = 1200; // "Голос" органа
filter.Q.value = 3;
filter.gain.value = 12; // Подчеркиваем средние частоты
