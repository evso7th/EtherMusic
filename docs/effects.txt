// public/worklets/reverb-processor.js

class ReverbProcessor extends AudioWorkletProcessor {
  constructor() {
    super();

    // Голоса
    this.voices = Array(4).fill(null).map(() => ({
      isActive: false,
      frequency: 0,
      gain: 0,
      targetGain: 0,
      phase: 0
    }));

    // Реверб-цепь (одна на все голоса)
    this.delay = new DelayNode(this.context, { delayTime: 0.1 });
    this.filter = new BiquadFilterNode(this.context, {
      type: 'lowpass',
      frequency: 1000,
      Q: 1
    });
    this.feedbackGain = new GainNode(this.context, { gain: 0.65 });

    // Соединяем: delay → filter → feedbackGain → delay
    this.delay.connect(this.filter);
    this.filter.connect(this.feedbackGain);
    this.feedbackGain.connect(this.delay);

    // Вход: голоса → delay → output
    // Выход: прямой сигнал + реверб
  }

  handleMessage(event) {
    const { type, voiceId, frequency, volume } = event.data;
    if (type === 'noteOn') {
      const voice = this.voices[voiceId];
      if (voice) {
        voice.isActive = true;
        voice.frequency = frequency;
        voice.targetGain = volume;
        voice.gain = 0;
        voice.phase = 0;
      }
    }

    if (type === 'noteOff') {
      const voice = this.voices[voiceId];
      if (voice) voice.targetGain = 0;
    }
  }

  process(inputs, outputs, parameters) {
    const output = outputs[0];

    // Буфер для реверб-сигнала
    const reverbBuffer = new Float32Array(output[0].length);

    for (let channel = 0; channel < output.length; channel++) {
      for (let i = 0; i < output[channel].length; i++) {
        let direct = 0;

        // Генерация прямого сигнала
        for (const voice of this.voices) {
          if (voice.isActive) {
            if (voice.gain < voice.targetGain) voice.gain += 0.01;
            else if (voice.gain > voice.targetGain) voice.gain -= 0.01;

            voice.phase += (voice.frequency / sampleRate) * 2 * Math.PI;
            if (voice.phase >= 2 * Math.PI) voice.phase -= 2 * Math.PI;

            direct += Math.sin(voice.phase) * voice.gain;
          }
        }

        // Прямой сигнал в реверб-цепь
        this.delay.inputs[0][i] = direct * 0.4;

        // Чтение выхода реверб-цепи
        const reverbSample = this.delay.outputs[0][i];

        // Смешиваем: прямой + реверб
        output[channel][i] = (direct * 0.7) + (reverbSample * 0.5);
      }
    }

    return true;
  }
}

registerProcessor('reverb-processor', ReverbProcessor);

// public/worklets/reverb-processor.js

class ReverbProcessor extends AudioWorkletProcessor {
  constructor() {
    super();

    // Единая реверберационная цепь
    this.delay = new DelayNode(this.context, { delayTime: 0.1 });
    this.filter = new BiquadFilterNode(this.context, {
      type: 'lowpass',
      frequency: 1000,
      Q: 1
    });
    this.feedbackGain = new GainNode(this.context, { gain: 0.65 });

    // Соединяем: delay → filter → feedbackGain → delay
    this.delay.connect(this.filter);
    this.feedbackGain.connect(this.delay);
    this.filter.connect(this.feedbackGain);
  }

  process(inputs, outputs, parameters) {
    const input = inputs[0];  // Сигнал от send-каналов
    const output = outputs[0]; // Реверб-хвост

    if (!input || !output) return true;

    for (let channel = 0; channel < output.length; channel++) {
      for (let i = 0; i < output[channel].length; i++) {
        // Входной сигнал (от всех синтезаторов)
        const dry = input[channel][i];

        // Записываем в цепь реверба
        this.delay.inputs[0][i] = dry;

        // Читаем "мокрый" сигнал
        const wet = this.delay.outputs[0][i];

        // Возвращаем реверб
        output[channel][i] = wet * 0.6;
      }
    }

    return true;
  }
}

registerProcessor('reverb-processor', ReverbProcessor);

class DistortionProcessor extends AudioWorkletProcessor {
  constructor() {
    super();
    this.drive = 1.0; // Уровень искажения
    this.color = 0.5;  // Цвет искажения (смешивание с чистым сигналом)
  }

  // Функция искажения: waveshaper
  waveshaper(sample, drive) {
    // Усиливаем сигнал
    sample *= drive;

    // Нелинейная функция искажения (soft clipping)
    if (sample > 1) {
      return 1;
    } else if (sample < -1) {
      return -1;
    } else {
      // Кубическая кривая — мягкие искажения
      return 1.5 * sample - 0.5 * sample * sample * sample;
    }
  }

  process(inputs, outputs, parameters) {
    const input = inputs[0];
    const output = outputs[0];

    if (!input || !output) return true;

    for (let channel = 0; channel < output.length; channel++) {
      for (let i = 0; i < output[channel].length; i++) {
        let sample = input[channel][i];

        // Применяем искажение
        const distorted = this.waveshaper(sample, this.drive);

        // Смешиваем с чистым сигналом (color control)
        const mixed = (distorted * this.color) + (sample * (1 - this.color));

        output[channel][i] = mixed;
      }
    }

    return true;
  }
}

registerProcessor('distortion-processor', DistortionProcessor);

// Без ворклета, в основном потоке
const createDistortion = (context: AudioContext, amount = 50) => {
  const ws = context.createWaveShaper();
  const samples = 4096;
  const curve = new Float32Array(samples);

  for (let i = 0; i < samples; i++) {
    const x = (i * 2) / samples - 1;
    // Формула мягкого клиппинга
    curve[i] = (Math.PI + amount) * x / (Math.PI + amount * Math.abs(x));
  }

  ws.curve = curve;
  ws.oversample = '4x'; // Уменьшает артефакты

  return ws;
};

// В AudioWorkletProcessor
process(inputs, outputs) {
  const input = inputs[0];
  const output = outputs[0];

  const bitDepth = 8;  // 8-bit
  const sampleRateReduction = 4;

  for (let i = 0; i < input[0].length; i++) {
    if (i % sampleRateReduction === 0) {
      const sample = input[0][i];
      const crushed = Math.round(sample * (2 ** bitDepth)) / (2 ** bitDepth);
      output[0][i] = crushed;
    } else {
      output[0][i] = output[0][i-1]; // Повтор предыдущего
    }
  }
  return true;
}
